cmake_minimum_required(VERSION 3.24)
project(GauntletMovementCapture LANGUAGES CXX)
find_package(Modlock CONFIG REQUIRED)
add_library(movement_capture MODULE observer_plugin.cc)
set_target_properties(movement_capture PROPERTIES PREFIX "")
target_link_libraries(movement_capture PRIVATE modlock::sdk)
target_compile_features(movement_capture PRIVATE cxx_std_23)
install(TARGETS movement_capture RUNTIME DESTINATION bin LIBRARY DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  add_executable(capture_core_test recorder_core_test.cc)
  target_compile_features(capture_core_test PRIVATE cxx_std_20)
  add_test(NAME capture_core COMMAND capture_core_test)
endif()
